//HashTables (HashTables_3)
 bool remove(std::string key)
    {
        int index = hash(key);
        bool removed = false;

        for (auto it = data[index].begin(); it != data[index].end() && !removed; ++it)
        {
            if (it->key == key)
            {
                data[index].erase(it);
                removed = true;
            }
        }

        return removed;
    }

// удаления яруса листьев
btree::ptrNODE removeLeaves(btree::ptrNODE root) 
{
	if (root == nullptr) {
		return nullptr;
	}
	if (root->left == nullptr && root->right == nullptr) {
		delete root; 
		return nullptr; 
	}

	root->left = removeLeaves(root->left);
	root->right = removeLeaves(root->right);

	return root;
}

//суммы элементов от корня до заданного уровня

int sum_fromroot(btree::ptrNODE root, int targetlevel)
{
	int result{};
	int culevel{};
	std::queue < std::pair<btree::ptrNODE, int>> st;
	st.push({ root,0 });
	while (!st.empty())
	{
		root = st.front().first;
		culevel = st.front().second;
		st.pop();

		if (targetlevel >= culevel)
			result += root->info;
		if (root->left)
			st.push({ root->left, culevel + 1 });
		if (root->right)
			st.push({ root->right, culevel + 1 });
	}
	return result;
}

void doubleLastChar(ttree::TTREE& tree, ttree::ptrNODE& root)
{
	if (root)
	{
		for (int i = 0; i < 26; ++i)
		{
			if (root->ptrs[i])
			{
				if (root->ptrs[i]->eow)
				{
					root->ptrs[i]->ptrs[i] = new ttree::NODE();
					root->ptrs[i]->eow = false;
					root->ptrs[i]->ptrs[i]->eow = true;

				}
				else
					doubleLastChar(tree, root->ptrs[i]);
			}
		}
	}
}

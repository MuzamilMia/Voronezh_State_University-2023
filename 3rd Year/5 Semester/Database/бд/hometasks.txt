SELECT 
    CONCAT(lastName, ' ', firstName, ' ', middleName) as fullName,
    CASE 
        WHEN role = 'teacher' THEN CAST(subjectCount as VARCHAR(10))
        ELSE CAST(goodGrades as VARCHAR(10))
    END as info
FROM (
    SELECT 
        t.lastName, t.firstName, t.middleName,
        'teacher' as role,
        COUNT(DISTINCT td.disciplineId) as subjectCount,
        0 as goodGrades
    FROM teachers t
    LEFT JOIN teacher_disciplines td ON t.teacherId = td.teacherId
    GROUP BY t.teacherId, t.lastName, t.firstName, t.middleName
    
    UNION ALL
    
    SELECT 
        s.lastName, s.firstName, s.middleName,
        'student' as role,
        0 as subjectCount,
        COUNT(DISTINCT CASE 
            WHEN p.grade IN (4, 5) THEN p.disciplineId 
        END) as goodGrades
    FROM students s
    LEFT JOIN performances p ON s.studentId = p.studentId
    GROUP BY s.studentId, s.lastName, s.firstName, s.middleName
